# Followers & Votes

Manage followers and votes for topic entities.

## Followers

### Add Follower

`PUT /v1/topics/{id}/followers`

<ParamField path="id" type="string" required>
  UUID of the topic.
</ParamField>

<ParamField body="id" type="string" required>
  UUID of the user to add as a follower (sent as request body string).
</ParamField>

### Remove Follower

`DELETE /v1/topics/{id}/followers/{userId}`

<ParamField path="id" type="string" required>
  UUID of the topic.
</ParamField>

<ParamField path="userId" type="string" required>
  UUID of the user to remove.
</ParamField>

## Votes

### Add or Update Vote

`PUT /v1/topics/{id}/vote`

<ParamField path="id" type="string" required>
  UUID of the topic.
</ParamField>

<ParamField body="updatedVoteType" type="string" required>
  Vote type: `votedUp`, `votedDown`, or `unVoted`.
</ParamField>

from metadata.generated.schema.entity.services.connections.metadata.openMetadataConnection import (

```java PUT /v1/topics/{id}/followers
import java.util.Map;
import java.util.UUID;
import org.openmetadata.sdk.network.HttpClient;
import org.openmetadata.sdk.network.HttpMethod;

// client is an initialized OpenMetadataClient.

HttpClient http = client.getHttpClient();

String topicId = "0a021819-982e-4ee4-a5a2-4af30eaa3016";
UUID userId = UUID.fromString("b6f7a1e2-3c4d-4e5f-8a9b-0c1d2e3f4a5b");

// Add a follower
http.executeForString(HttpMethod.PUT, "/v1/topics/" + topicId + "/followers", userId);

// Remove a follower
http.executeForString(HttpMethod.DELETE, "/v1/topics/" + topicId + "/followers/" + userId, null);

// Add a vote
http.executeForString(HttpMethod.PUT, "/v1/topics/" + topicId + "/vote", Map.of("updatedVoteType", "votedUp"));

// Remove a vote
http.executeForString(HttpMethod.PUT, "/v1/topics/" + topicId + "/vote", Map.of("updatedVoteType", "unVoted"));
```
