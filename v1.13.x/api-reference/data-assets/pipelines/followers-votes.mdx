# Followers & Votes

Manage followers and votes for pipeline entities.

## Followers

### Add Follower

`PUT /v1/pipelines/{id}/followers`

<ParamField path="id" type="string" required>
  UUID of the pipeline.
</ParamField>

<ParamField body="id" type="string" required>
  UUID of the user to add as a follower (sent as request body string).
</ParamField>

### Remove Follower

`DELETE /v1/pipelines/{id}/followers/{userId}`

<ParamField path="id" type="string" required>
  UUID of the pipeline.
</ParamField>

<ParamField path="userId" type="string" required>
  UUID of the user to remove.
</ParamField>

## Votes

### Add or Update Vote

`PUT /v1/pipelines/{id}/vote`

<ParamField path="id" type="string" required>
  UUID of the pipeline.
</ParamField>

<ParamField body="updatedVoteType" type="string" required>
  Vote type: `votedUp`, `votedDown`, or `unVoted`.
</ParamField>

from metadata.sdk import client, configure

configure(
    host="https://your-company.open-metadata.org/api",
    jwt_token="your-jwt-token"

```java PUT /v1/pipelines/{id}/followers
import java.util.Map;
import java.util.UUID;
import org.openmetadata.sdk.network.HttpClient;
import org.openmetadata.sdk.network.HttpMethod;

HttpClient http = client.getHttpClient();

String pipelineId = "538faa63-d204-46ff-aead-d158d0401cac";
UUID userId = UUID.fromString("b6f7a1e2-3c4d-4e5f-8a9b-0c1d2e3f4a5b");

// Add a follower
http.executeForString(HttpMethod.PUT, "/v1/pipelines/" + pipelineId + "/followers", userId);

// Remove a follower
http.executeForString(HttpMethod.DELETE, "/v1/pipelines/" + pipelineId + "/followers/" + userId, null);

// Add a vote
http.executeForString(HttpMethod.PUT, "/v1/pipelines/" + pipelineId + "/vote", Map.of("updatedVoteType", "votedUp"));

// Remove a vote
http.executeForString(HttpMethod.PUT, "/v1/pipelines/" + pipelineId + "/vote", Map.of("updatedVoteType", "unVoted"));
```
